local function trocarAba(nomeAba)
    for nome, dados in pairs(Abas) do
        if nome == nomeAba then
            dados.frame.Visible = true
            dados.botao.BackgroundColor3 = COR_ABA_ON
            dados.botao.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            dados.frame.Visible = false
            dados.botao.BackgroundColor3 = COR_ABA_OFF
            dados.botao.TextColor3 = COR_TXT_OFF
        end
    end
    ContentTitle.Text = nomeAba
    AbaAtual = nomeAba
end

    abaFrame.Size = UDim2.new(1, 0, 0, 0)
    abaFrame.BackgroundTransparency = 1

    abaFrame.Parent = FunctionList

    local abaLayout = Instance.new("UIListLayout")
    abaLayout.Padding = UDim.new(0, 7)
    abaLayout.SortOrder = Enum.SortOrder.LayoutOrder
    abaLayout.Parent = abaFrame

    abaLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        abaFrame.Size = UDim2.new(1, 0, 0, abaLayout.AbsoluteContentSize.Y + 10)
    end)

        trocarAba(nome)
    end)

    Abas[nome] = { botao = botao, frame = abaFrame }
    if not AbaAtual then trocarAba(nome) end

local function criarToggle(parent, texto, ordem, callback)

    circle.Parent = bg
    Instance.new("UICorner", circle).CornerRadius = UDim.new(1, 0)

    local click = Instance.new("TextButton")
    click.Size = UDim2.new(1, 0, 1, 0)
    click.BackgroundTransparency = 1

    local estado = false
    click.MouseButton1Click:Connect(function()

            bg.BackgroundColor3 = COR_TOG_ON

            bg.BackgroundColor3 = COR_TOG_OFF

criarToggle(AbaP, "Focus", 2, function(ativo)
    if ativo then
        task.spawn(function()
            local angle = 0
            while _G.Aether_Focus do
                local char = LP.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if root then
                    local best, bestD = nil, math.huge
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p ~= LP and p.Character then
                            local tr = p.Character:FindFirstChild("HumanoidRootPart")
                            local hum = p.Character:FindFirstChildOfClass("Humanoid")
                            if tr and hum and hum.Health > 0 then
                                local d = (tr.Position - root.Position).Magnitude
                                if d < bestD then bestD = d; best = tr end
                            end
                        end
                    end
                    if best then
                        angle = angle + 0.1
                        local offset = Vector3.new(math.cos(angle) * 6, 3, math.sin(angle) * 6)
                        local hum = char:FindFirstChildOfClass("Humanoid")
                        if hum then hum.PlatformStand = true end
                        root.CFrame = CFrame.new(best.Position + offset, best.Position)
                    end
                end
                RunService.Heartbeat:Wait()
            end
            local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
            if hum then hum.PlatformStand = false end
        end)
        _G.Aether_Focus = true
    else
        _G.Aether_Focus = false
    end
end)

        noclipAtivo = ativo

            task.spawn(function()
                while noclipAtivo do
                    local c = LP.Character
                    if c then
                        for _, p in ipairs(c:GetDescendants()) do
                            if p:IsA("BasePart") then p.CanCollide = false end
                        end
                    end
                    task.wait(0.2)
                end
            end)
        end
    end)

        if infJumpAtivo then
            local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
            if h then h:ChangeState(Enum.HumanoidStateType.Jumping) end
        end
    end)
